cmake_minimum_required(VERSION 3.16)
project(simpleson_assignment LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Select one implementation per build directory; never mix its headers and library.
set(SIMPLESON_VARIANT "optimized" CACHE STRING "original, reference or optimized")
set_property(CACHE SIMPLESON_VARIANT PROPERTY STRINGS original reference optimized)
if(SIMPLESON_VARIANT STREQUAL "original")
	set(JSON_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/baseline")
elseif(SIMPLESON_VARIANT STREQUAL "reference")
	set(JSON_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/reference")
elseif(SIMPLESON_VARIANT STREQUAL "optimized")
	set(JSON_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
else()
	message(FATAL_ERROR "Unknown SIMPLESON_VARIANT: ${SIMPLESON_VARIANT}")
endif()

add_library(simpleson "${JSON_SOURCE_DIR}/json.cpp")
target_include_directories(simpleson PUBLIC "${JSON_SOURCE_DIR}")
target_compile_features(simpleson PUBLIC cxx_std_17)
if(MSVC)
	add_compile_options(/W4 /WX /utf-8)
	target_compile_options(simpleson PRIVATE /W4 /WX /utf-8)
	add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
	target_compile_definitions(simpleson PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
	add_compile_options(-Wall -Wextra -Werror)
	target_compile_options(simpleson PRIVATE -Wall -Wextra -Werror)
endif()

enable_testing()
add_subdirectory(task01)
add_subdirectory(task02)
add_executable(simpleson_behavior checks/behavior.cpp)
target_link_libraries(simpleson_behavior PRIVATE simpleson)
add_executable(simpleson_reference_measure checks/reference_measure.cpp alloccounter/alloc_counter.cpp)
target_include_directories(simpleson_reference_measure PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/alloccounter")
target_link_libraries(simpleson_reference_measure PRIVATE simpleson)

add_test(NAME assignment_demo COMMAND simpleson_task01)
add_test(NAME assignment_memory COMMAND simpleson_task02)
# The complete suite also checks storage improvements absent from the original.
if(SIMPLESON_VARIANT STREQUAL "optimized")
	add_subdirectory(test)
	add_subdirectory(issues)
	add_subdirectory(examples)
endif()

add_subdirectory(task03)

set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT simpleson_task01)
